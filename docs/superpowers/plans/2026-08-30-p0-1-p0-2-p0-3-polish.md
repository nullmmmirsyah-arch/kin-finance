```ts
function matchesSearch(row: Doc<"transactions">, search: string, hydrated?: {category?: Doc<"categories">, account?: Doc<"accounts">, toAccount?: Doc<"accounts">}): boolean {
  const note = (row.note ?? "").toLowerCase();
  if (note.includes(search)) return true;
  // amount: match abs amount as decimal string without commas, also with commas stripped search
  const amountStr = String(Math.abs(row.amount));
  const searchDigits = search.replace(/,/g,"");
  if (amountStr.includes(searchDigits) || String(row.amount).includes(searchDigits)) return true;
  // hydrated names
  if (hydrated?.account && hydrated.account.name.toLowerCase().includes(search)) return true;
  if (hydrated?.toAccount && hydrated.toAccount.name.toLowerCase().includes(search)) return true;
  if (hydrated?.category && hydrated.category.name.toLowerCase().includes(search)) return true;
  return false;
}
```

In `utils/format.ts` add:

```ts
export function wasDecimalTruncated(raw: string): boolean {
  return raw.includes(".");
}
export function detectAmountTruncation(raw: string, formatted: string): { truncated: boolean; reason: "decimal" | "nonDigit" | null } {
  if (raw.includes(".")) return { truncated: true, reason: "decimal" };
  const rawDigits = raw.replace(/[^0-9]/g,"");
  const formattedDigits = formatted.replace(/[^0-9]/g,"");
  if (rawDigits !== formattedDigits) return { truncated: true, reason: "nonDigit" };
  if (raw !== formatted && raw.length !== formatted.length) return { truncated: true, reason: "nonDigit" };
  return { truncated: false, reason: null };
}
```

Keep existing `formatAmountInput` unchanged.
